## Task 3：InstrumentFaultRateCalculator（核心，复用 outlier_reasons）

**新增文件**: `backend/app/services/metric_calculator/instrument_fault.py`

```python
"""仪表故障率计算器（Phase 1）.

复用既有 8 类异常检测（preprocessing/outlier_detection.py）的检测结果，
聚合 PV 信号上 OUT_OF_RANGE/FROZEN/JUMP 三类原因码为故障率指标。

公式：故障率 = 故障点数 / 总点数 × 100%
定位：辅助诊断指标，AGGREGATABLE（装置级故障率有意义）。
"""

from __future__ import annotations

from app.contracts.data_types import MetricDataBundle, MetricResult, OutlierReason
from app.services.metric_calculator.base import MetricCalculatorBase

#: 仪表故障对应的异常原因码集合（HiaMonitor 超限/冻结/突变 三类）
_FAULT_REASONS = frozenset({
    OutlierReason.OUT_OF_RANGE.value,
    OutlierReason.FROZEN.value,
    OutlierReason.JUMP.value,
})


class InstrumentFaultRateCalculator(MetricCalculatorBase):
    """仪表故障率（复用 DataBlock.outlier_reasons，不新增预处理步骤）."""

    @property
    def metric_code(self) -> str:
        return "instrument_fault_rate"

    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        block = bundle.data_block
        n = block.point_count
        if n <= 0:
            return self._make_inconclusive(bundle, "empty_data_block")

        # 从 DataBlock.outlier_reasons 读取 PV 每点原因码（非 bundle.signals）
        # outlier_reasons["pv"] 为密集 list[list[str]]，索引对齐数据点
        pv_reasons: list[list[str]] = block.outlier_reasons.get("pv", [])
        # 防御：若为稀疏结构，按长度补齐
        if len(pv_reasons) < n:
            pv_reasons = pv_reasons + [[] for _ in range(n - len(pv_reasons))]

        freeze_cnt = mutation_cnt = overrange_cnt = fault_pts = 0
        for reasons in pv_reasons:
            rset = set(reasons) if reasons else set()
            hit = False
            if OutlierReason.FROZEN.value in rset:
                freeze_cnt += 1; hit = True
            if OutlierReason.JUMP.value in rset:
                mutation_cnt += 1; hit = True
            if OutlierReason.OUT_OF_RANGE.value in rset:
                overrange_cnt += 1; hit = True
            if hit:
                fault_pts += 1

        fault_rate = self._clamp((fault_pts / n) * 100.0)
        # valid_rate 用 mask 有效率（可信度判定用），故障率本身用全量点
        return self._make_result(
            bundle,
            fault_rate,
            {
                "fault_rate": self._round2(fault_rate),
                "freeze_count": freeze_cnt,
                "mutation_count": mutation_cnt,
                "overrange_count": overrange_cnt,
                "fault_point_count": fault_pts,
                "sample_count": n,
                "source": "outlier_reasons",
            },
        )
```

**关键说明**：
- 读 `bundle.data_block.outlier_reasons["pv"]`（**非** `bundle.signals`，评审 P0-1）
- 故障率用全量点 `block.point_count`，**不**用 `masked_indices`（故障点 valid=False 已被排除在 mask 外，用 mask 会漏掉所有故障点）
- `_make_result` 内部用 `_get_valid_rate`（mask 率）定可信度——合理：数据有效率低则可信度低
- **不修改** [outlier_detection.py](file:///Users/zhangping/DEV/CLPM/backend/app/services/preprocessing/outlier_detection.py)

### 4.1 统计指标（statistics.py）

```python
"""PV/SP/OP/偏差 统计指标（均值+标准差），Phase 1。

定位：DISPLAY_ONLY（不参与节点聚合，避免均值再平均失真）。
复用 base._get_masked_values 提取掩码后有效值，用 statistics 计算。
"""
from __future__ import annotations
import statistics as stats
from app.contracts.data_types import MetricDataBundle, MetricResult
from app.services.metric_calculator.base import MetricCalculatorBase


class _MeanBase(MetricCalculatorBase):
    """均值基类：子类指定 _tag。"""
    _tag: str = ""

    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        vals = [float(v) for v in self._get_masked_values(bundle, self._tag) if v is not None]
        if len(vals) < 2:
            return self._make_inconclusive(bundle, "insufficient_points")
        mean = stats.mean(vals)
        return self._make_result(bundle, mean, {"mean": self._round2(mean), "n": len(vals)})


class _StdBase(MetricCalculatorBase):
    """标准差基类：子类指定 _tag。"""
    _tag: str = ""

    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        vals = [float(v) for v in self._get_masked_values(bundle, self._tag) if v is not None]
        if len(vals) < 2:
            return self._make_inconclusive(bundle, "insufficient_points")
        std = stats.pstdev(vals)  # 总体标准差
        return self._make_result(bundle, std, {"std": self._round2(std), "n": len(vals)})


class PvMeanCalculator(_MeanBase):
    @property
    def metric_code(self) -> str: return "pv_mean"
    _tag = "pv"

class PvStdCalculator(_StdBase):
    @property
    def metric_code(self) -> str: return "pv_std"
    _tag = "pv"

# 同模式：SpMean/SpStd(_tag="sp")、OpMean/OpStd(_tag="op")
# ErrorMean/ErrorStd：用 _get_masked_pair(bundle,"pv","sp") 算 (pv-sp)
class ErrorMeanCalculator(MetricCalculatorBase):
    @property
    def metric_code(self) -> str: return "error_mean"
    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        pairs = self._get_masked_pair(bundle, "pv", "sp")
        errs = [float(p - s) for p, s in pairs if p is not None and s is not None]
        if len(errs) < 2:
            return self._make_inconclusive(bundle, "insufficient_points")
        return self._make_result(bundle, stats.mean(errs), {"mean": self._round2(stats.mean(errs)), "n": len(errs)})
# ErrorStdCalculator 同理用 stats.pstdev(errs)
```

### 4.2 阀门诊断指标（valve_diagnosis.py）

```python
"""阀门线性度/非线性度/运行区间，Phase 1。

PV-OP 线性相关系数反映阀门线性特性；运行区间为 OP 的 [min,max]。
定位：DISPLAY_ONLY。
"""
from __future__ import annotations
import math
from app.contracts.data_types import MetricDataBundle, MetricResult
from app.services.metric_calculator.base import MetricCalculatorBase


class ValveLinearityCalculator(MetricCalculatorBase):
    @property
    def metric_code(self) -> str: return "valve_linearity"
    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        pairs = self._get_masked_pair(bundle, "pv", "op")
        xs = [float(p) for p, _ in pairs if p is not None]
        ys = [float(o) for _, o in pairs if o is not None]
        if len(xs) < 3:
            return self._make_inconclusive(bundle, "insufficient_points")
        # 皮尔逊相关系数绝对值
        r = self._pearson(xs, ys)
        linearity = abs(r) if r is not None else 0.0
        return self._make_result(bundle, linearity, {"r": round(r or 0, 4), "n": len(xs)})

    @staticmethod
    def _pearson(xs: list[float], ys: list[float]) -> float | None:
        n = len(xs); mx = sum(xs)/n; my = sum(ys)/n
        sxy = sum((x-mx)*(y-my) for x,y in zip(xs,ys))
        sxx = sum((x-mx)**2 for x in xs); syy = sum((y-my)**2 for y in ys)
        denom = math.sqrt(sxx*syy)
        return sxy/denom if denom else None


class ValveNonlinearityCalculator(MetricCalculatorBase):
    @property
    def metric_code(self) -> str: return "valve_nonlinearity"
    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        pairs = self._get_masked_pair(bundle, "pv", "op")
        xs = [float(p) for p, _ in pairs if p is not None]
        if len(xs) < 3:
            return self._make_inconclusive(bundle, "insufficient_points")
        r = ValveLinearityCalculator._pearson(xs, [float(o) for _,o in pairs if o is not None])
        nonlinearity = 1.0 - (abs(r) if r is not None else 0.0)
        return self._make_result(bundle, nonlinearity, {"nonlinearity": round(nonlinearity, 4)})


class ValveOperatingRangeCalculator(MetricCalculatorBase):
    @property
    def metric_code(self) -> str: return "valve_operating_range"
    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        ops = [float(v) for v in self._get_masked_values(bundle, "op") if v is not None]
        if len(ops) < 2:
            return self._make_inconclusive(bundle, "insufficient_points")
        op_min, op_max = min(ops), max(ops)
        # value 用区间跨度，min/max 存 details
        return self._make_result(
            bundle, op_max - op_min,
            {"op_min": self._round2(op_min), "op_max": self._round2(op_max), "span": self._round2(op_max-op_min)},
        )
```

### 4.3 设定点穿越次数 + 振荡幅值

```python
"""设定点穿越次数 + 振荡幅值，Phase 1。

穿越次数：PV 穿越 SP 的次数（含简单去抖）。
振荡幅值：PV 偏离 SP 的平均绝对偏差。
"""
from __future__ import annotations
from app.contracts.data_types import MetricDataBundle, MetricResult
from app.services.metric_calculator.base import MetricCalculatorBase


class SetpointCrossingCountCalculator(MetricCalculatorBase):
    @property
    def metric_code(self) -> str: return "setpoint_crossing_count"
    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        pairs = self._get_masked_pair(bundle, "pv", "sp")
        if len(pairs) < 3:
            return self._make_inconclusive(bundle, "insufficient_points")
        diffs = [float(p) - float(s) for p, s in pairs]
        crossings = sum(1 for i in range(1, len(diffs)) if diffs[i-1] * diffs[i] < 0)
        # MetricResult.value 为 float，穿越次数为整数语义
        return self._make_result(bundle, float(crossings), {"crossing_count": crossings, "n": len(diffs)})


class OscillationAmplitudeCalculator(MetricCalculatorBase):
    """振荡幅值（L2，依赖 oscillation_rate）。"""
    @property
    def metric_code(self) -> str: return "oscillation_amplitude"
    def calculate(self, bundle: MetricDataBundle) -> MetricResult:
        pairs = self._get_masked_pair(bundle, "pv", "sp")
        if len(pairs) < 3:
            return self._make_inconclusive(bundle, "insufficient_points")
        abs_errs = [abs(float(p) - float(s)) for p, s in pairs]
        amp = sum(abs_errs) / len(abs_errs)
        return self._make_result(bundle, amp, {"amplitude": self._round2(amp), "n": len(abs_errs)})
```

"""Phase 1 新增指标单元测试。"""
import pytest
from app.contracts.data_types import (
    DataBlock, DataLineage, MetricDataBundle, OutlierReason, QualitySummary,

# ... 其余 import


def _make_bundle(signals, outlier_reasons=None, n=10):
    """构造测试 bundle：全 valid，masked_indices=全点。"""
    block = DataBlock(
        data_block_id="db_test_BASE_1s", loop_id="L1", tag_group="BASE",
        sampling_freq="1s", timestamps=list(range(n)), signals=signals,
        validity={f"{k}_valid": [True]*n for k in signals},
        outlier_reasons=outlier_reasons or {},
        quality_summary=QualitySummary(total_count=n, valid_count=n, valid_rate=1.0),
        point_count=n,
    )
    return MetricDataBundle(
        metric_code="x", data_block=block, mask_expression="pv_valid",
        masked_indices=list(range(n)), lineage=DataLineage(valid_rate=1.0),
    )


class TestInstrumentFaultRate:
    def test_zero_fault(self):
        b = _make_bundle({"pv": [50.0]*10}, outlier_reasons={"pv": [[]]*10})
        r = InstrumentFaultRateCalculator().calculate(b)
        assert r.value == 0.0
        assert r.details["fault_point_count"] == 0

    def test_all_frozen(self):
        reasons = {"pv": [[OutlierReason.FROZEN.value]]*10}
        b = _make_bundle({"pv": [50.0]*10}, outlier_reasons=reasons)
        r = InstrumentFaultRateCalculator().calculate(b)
        assert r.value == 100.0
        assert r.details["freeze_count"] == 10

    def test_mixed_faults(self):
        reasons = {"pv": [
            [OutlierReason.OUT_OF_RANGE.value], [], [OutlierReason.JUMP.value],
            [], [], [OutlierReason.FROZEN.value], [], [], [], [],
        ]}
        b = _make_bundle({"pv": list(range(10))}, outlier_reasons=reasons)
        r = InstrumentFaultRateCalculator().calculate(b)
        assert r.value == 30.0  # 3/10
        assert r.details["overrange_count"] == 1
        assert r.details["mutation_count"] == 1
        assert r.details["freeze_count"] == 1

    def test_empty_block_inconclusive(self):
        b = _make_bundle({"pv": []}, n=0)
        r = InstrumentFaultRateCalculator().calculate(b)
        assert r.value is None  # INCONCLUSIVE
