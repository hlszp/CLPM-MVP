## 2. Linux root 整块

```bash
set -e
printf '%s  %s\n' '257a5e24db1438570ea6adcb094b0feecd0d4e14e3fa900c133cf646f9bd0864' '/tmp/clpm-src-1050.tar.gz' | sha256sum -c -
rm -rf /tmp/clpm-src-1050 && mkdir -p /tmp/clpm-src-1050
tar -xzf /tmp/clpm-src-1050.tar.gz -C /tmp/clpm-src-1050
cd /tmp/clpm-src-1050
BASE_BE=$(docker images --format '{{.Repository}}:{{.Tag}}' | grep '^clpm-backend:' | sort | tail -1)
BASE_FE=$(docker images --format '{{.Repository}}:{{.Tag}}' | grep '^clpm-frontend:' | sort | tail -1)
echo "BASE_BE=$BASE_BE BASE_FE=$BASE_FE"
docker build -f Dockerfile.backend --build-arg BASE="$BASE_BE" --build-arg SOURCE_REVISION=1050 -t clpm-backend:full-1050 .
docker build -f Dockerfile.frontend --build-arg BASE="$BASE_FE" --build-arg SOURCE_REVISION=1050 -t clpm-frontend:full-1050 .
cd /opt/clpm-delivery-20261005-full
cp -p --update=none docker-compose.prod.yml docker-compose.prod.yml.bak-1050
sed -i '/image: clpm-backend:/s/clpm-backend:[^ ]*/clpm-backend:full-1050/' docker-compose.prod.yml
sed -i '/image: clpm-frontend:/s/clpm-frontend:[^ ]*/clpm-frontend:full-1050/' docker-compose.prod.yml
if grep -q '^APP_VERSION=' .env.prod; then sed -i 's/^APP_VERSION=.*/APP_VERSION=v7.5.6-1050-full/' .env.prod; fi
docker compose --env-file .env.prod -f docker-compose.prod.yml -f docker-compose.shadow-1046.yml --profile tdengine up -d --force-recreate backend celery-worker celery-beat frontend
sleep 8
docker ps --format '{{.Names}}\t{{.Status}}' | grep clpm
docker exec clpm-backend python -m scripts.ten_minute_shadow status
```
