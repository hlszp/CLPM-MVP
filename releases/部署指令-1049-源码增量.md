## 2. Linux root：解压 + 现场构建 + 切换

```bash
set -e
printf '%s  %s\n' '7275c31a8f09054c6f18b50ab7955b68848d94d39252763a4813bb143ae60661' '/tmp/clpm-src-1049.tar.gz' | sha256sum -c -
rm -rf /tmp/clpm-src-1049 && mkdir -p /tmp/clpm-src-1049
tar -xzf /tmp/clpm-src-1049.tar.gz -C /tmp/clpm-src-1049
cd /tmp/clpm-src-1049
# 现场 BASE 取当前运行版本（若已是 1047/1048/1049 均可）
BASE_BE=$(docker images --format '{{.Repository}}:{{.Tag}}' | grep '^clpm-backend:' | sort | tail -1)
BASE_FE=$(docker images --format '{{.Repository}}:{{.Tag}}' | grep '^clpm-frontend:' | sort | tail -1)
echo "BASE_BE=$BASE_BE BASE_FE=$BASE_FE"
docker build -f Dockerfile.backend --build-arg BASE="$BASE_BE" --build-arg SOURCE_REVISION=1049 -t clpm-backend:full-1049 .
docker build -f Dockerfile.frontend --build-arg BASE="$BASE_FE" --build-arg SOURCE_REVISION=1049 -t clpm-frontend:full-1049 .
cd /opt/clpm-delivery-20261005-full
cp -p --update=none docker-compose.prod.yml docker-compose.prod.yml.bak-1049src
sed -i '/image: clpm-backend:/s/clpm-backend:[^ ]*/clpm-backend:full-1049/' docker-compose.prod.yml
sed -i '/image: clpm-frontend:/s/clpm-frontend:[^ ]*/clpm-frontend:full-1049/' docker-compose.prod.yml
if grep -q '^APP_VERSION=' .env.prod; then sed -i 's/^APP_VERSION=.*/APP_VERSION=v7.5.6-1049-full/' .env.prod; fi
docker compose --env-file .env.prod -f docker-compose.prod.yml -f docker-compose.shadow-1046.yml --profile tdengine up -d --force-recreate backend celery-worker celery-beat frontend
sleep 8
docker ps --format '{{.Names}}\t{{.Status}}' | grep clpm
docker exec clpm-backend python -m scripts.ten_minute_shadow status
```
